import random class Player: def __init__(self, name, money=100000): self.name = name self.money = money self.position = 0 self.properties = [] # 보유한 도시 객체 목록 def can_afford(self, amount): return self.money >= amount def get_total_assets_value(self, board): """현금 + 보유 도시의 매각 가치(총 투자비용의 0.75배) 계산""" sellable_value = sum(int(prop.total_investment * 0.75) for prop in self.properties) return self.money + sellable_value def sell_property_interactive(self, needed_amount): """현금이 부족할 때 플레이어가 선택해서 자산을 매도하는 로직""" while self.money < needed_amount: if not self.properties: print(f"[{self.name}] 더 이상 매도할 자산이 없습니다.") return False print(f"\n⚠️ [자금 부족] 필요 금액: {needed_amount:,}원 | 현재 현금: {self.money:,}원") print("매도할 자산을 선택하세요:") for idx, prop in enumerate(self.properties): sell_price = int(prop.total_investment * 0.75) print(f" [{idx + 1}] {prop.name} (누적 투자: {prop.total_investment:,}원 -> 매각가: {sell_price:,}원)") try: choice = int(input("매도할 번호 입력 (취소 불가): ")) - 1 if 0 <= choice < len(self.properties): target_prop = self.properties.pop(choice) sell_price = int(target_prop.total_investment * 0.75) self.money += sell_price target_prop.reset() print(f"➔ {target_prop.name} 매도 완료! (+{sell_price:,}원 수령 | 현재 현금: {self.money:,}원)") else: print("잘못된 번호입니다. 다시 선택해주세요.") except ValueError: print("숫자를 입력해주세요.") return True class City: def __init__(self, name, land_cost, building_cost): self.name = name self.land_cost = land_cost self.building_cost = building_cost self.owner = None self.buildings = {"land": False, "building": False, "landmark": False} self.total_investment = 0 def reset(self): """매도 또는 인수 시 초기화""" self.owner = None self.buildings = {"land": False, "building": False, "landmark": False} self.total_investment = 0 def get_toll(self): """통행료 계산 (땅 10%, 건물 20%, 랜드마크 50%)""" toll = 0 if self.buildings["land"]: toll += int(self.land_cost * 0.1) if self.buildings["building"]: toll += int(self.building_cost * 0.2) if self.buildings["landmark"]: toll += int(self.building_cost * 0.5) return toll def buy_or_upgrade(self, player): """신규 매수 및 업그레이드 처리""" if self.owner is None: cost = self.land_cost if player.money < cost: if not player.sell_property_interactive(cost): print("자금이 부족하여 매수할 수 없습니다.") return player.money -= cost self.owner = player self.buildings["land"] = True self.total_investment += cost player.properties.append(self) print(f"[{player.name}] {self.name} 토지 매수 완료! (-{cost:,}원)") # 보유 중인 경우 업그레이드 여부 확인 if self.owner == player: if not self.buildings["building"]: cost = self.building_cost if player.money >= cost and input(f"{self.name}에 건물을 지으시겠습니까? ({cost:,}원) [y/n]: ").lower() == 'y': player.money -= cost self.buildings["building"] = True self.total_investment += cost print(f"[{player.name}] {self.name} 건물 건설 완료!") elif not self.buildings["landmark"]: cost = self.building_cost * 2 if player.money >= cost and input(f"{self.name}에 랜드마크를 올리시겠습니까? ({cost:,}원) [y/n]: ").lower() == 'y': player.money -= cost self.buildings["landmark"] = True self.total_investment += cost print(f"[{player.name}] 🎉 {self.name} 랜드마크 건설 완료! (인수 불가 지역)") def takeover(self, player): """1.5배 인수 처리 로직""" takeover_cost = int(self.total_investment * 1.5) print(f"\n상대 도시 인수 가능! 필요 비용: {takeover_cost:,}원 (투자비용 {self.total_investment:,}원의 1.5배)") choice = input(f"[{player.name}] {self.name}을(를) 인수하시겠습니까? [y/n]: ").lower() if choice == 'y': # 현금 부족 시 사용자 매도 절차 수행 if player.money < takeover_cost: if not player.sell_property_interactive(takeover_cost): print("자산 매도 후에도 자금이 부족하여 인수를 포기합니다.") return # 인수 비용 지불 및 전 주인에게 대금 전달 player.money -= takeover_cost self.owner.money += takeover_cost print(f"➔ [{self.owner.name}] 님에게 인수 대금 {takeover_cost:,}원이 전달되었습니다.") # 소유권 이전 self.owner.properties.remove(self) self.owner = player player.properties.append(self) print(f"🎉 [{player.name}] 님이 {self.name}을(를) 인수했습니다!") # --- 게임 실행 및 턴 처리 예시 --- def process_turn(player, city): print(f"\n🎲 [{player.name}] 님이 {city.name}에 도착했습니다.") # 1. 빈 땅인 경우 if city.owner is None: city.buy_or_upgrade(player) # 2. 내 땅인 경우 (업그레이드) elif city.owner == player: city.buy_or_upgrade(player) # 3. 상대방 땅인 경우 else: toll = city.get_toll() print(f"🚨 [{city.owner.name}] 님의 구역입니다! 통행료: {toll:,}원") # 통행료 지불 (부족 시 선택 매도) if player.money < toll: if not player.sell_property_interactive(toll): print(f"💥 [{player.name}] 님이 파산했습니다!") return False player.money -= toll city.owner.money += toll print(f"통행료 지불 완료 (남은 현금: {player.money:,}원)") # 랜드마크 전 단계일 경우 1.5배 인수 기회 제공 if not city.buildings["landmark"]: city.takeover(player) else: print("🏰 랜드마크가 건설되어 있어 인수가 불가능합니다.") return True
